# scripts/data_gen.py
import pandas as pd
import numpy as np
import datetime as dt
import random
import os

# Ensure reproducibility
np.random.seed(42)
random.seed(42)

def gerar_dados_projetos(n_projetos=200):
    """
    Gera um conjunto de dados simulados para projetos de engenharia/arquitetura.
    
    Parameters:
    -----------
    n_projetos : int
        Número de projetos a serem gerados
        
    Returns:
    --------
    pd.DataFrame
        DataFrame com os dados dos projetos
    """
    # Datas
    data_inicio = dt.datetime(2023, 1, 1)
    data_fim = dt.datetime(2025, 7, 30)  # Using today's date
    dias_intervalo = (data_fim - data_inicio).days
    
    # Dados básicos dos projetos
    projetos = {
        'id_projeto': range(1, n_projetos + 1),
        'nome_projeto': [f'Projeto {i:03d}' for i in range(1, n_projetos + 1)],
        'data_inicio': [data_inicio + dt.timedelta(days=random.randint(0, dias_intervalo//2)) for _ in range(n_projetos)],
        'tipo_projeto': np.random.choice(['Residencial', 'Comercial', 'Industrial', 'Infraestrutura'], n_projetos, p=[0.4, 0.3, 0.2, 0.1]),
        'regiao': np.random.choice(['Norte', 'Sul', 'Leste', 'Oeste', 'Centro'], n_projetos),
        'complexidade': np.random.choice(['Baixa', 'Média', 'Alta', 'Muito Alta'], n_projetos, p=[0.2, 0.4, 0.3, 0.1]),
        'cliente_id': np.random.randint(1, 51, n_projetos)
    }
    
    # Criando DataFrame base
    df_projetos = pd.DataFrame(projetos)
    
    # Adicionando métricas baseadas no tipo e complexidade
    # Duração (em dias)
    complexidade_to_fator = {
        'Baixa': 30, 
        'Média': 90, 
        'Alta': 180, 
        'Muito Alta': 365
    }
    
    tipo_to_fator = {
        'Residencial': 0.8,
        'Comercial': 1.0,
        'Industrial': 1.5,
        'Infraestrutura': 2.0
    }
    
    # Calculando duração
    df_projetos['duracao_prevista'] = df_projetos.apply(
        lambda row: int(complexidade_to_fator[row['complexidade']] * tipo_to_fator[row['tipo_projeto']] * 
                        (0.8 + 0.4 * np.random.random())),
        axis=1
    )
    
    # Data de conclusão prevista
    df_projetos['data_conclusao_prevista'] = df_projetos.apply(
        lambda row: row['data_inicio'] + dt.timedelta(days=row['duracao_prevista']),
        axis=1
    )
    
    # Orçamento (em milhares de R$)
    orcamento_base = {
        'Residencial': 250,
        'Comercial': 500,
        'Industrial': 1000,
        'Infraestrutura': 2000
    }
    
    complexidade_fator = {
        'Baixa': 0.8,
        'Média': 1.0,
        'Alta': 1.5,
        'Muito Alta': 2.5
    }
    
    df_projetos['orcamento'] = df_projetos.apply(
        lambda row: int(orcamento_base[row['tipo_projeto']] * complexidade_fator[row['complexidade']] * 
                        (0.9 + 0.2 * np.random.random()) * 1000),  # em R$
        axis=1
    )
    
    # Status do projeto
    df_projetos['status'] = df_projetos.apply(
        lambda row: 'Concluído' if row['data_conclusao_prevista'] < dt.datetime.now() else 
                   ('Em Andamento' if row['data_inicio'] < dt.datetime.now() else 'Planejado'),
        axis=1
    )
    
    # Progresso (%)
    df_projetos['progresso'] = df_projetos.apply(
        lambda row: 100 if row['status'] == 'Concluído' else 
                   (np.random.randint(60, 95) if row['status'] == 'Em Andamento' else 
                    np.random.randint(5, 30)),
        axis=1
    )
    
    # Equipe (número de pessoas)
    df_projetos['tamanho_equipe'] = df_projetos.apply(
        lambda row: max(2, int(np.sqrt(row['orcamento']/10000) * (0.8 + 0.4 * np.random.random()))),
        axis=1
    )
    
    # Calculando métricas de desempenho
    
    # Atraso (dias)
    df_projetos['atraso'] = df_projetos.apply(
        lambda row: int(row['duracao_prevista'] * (0.0 if np.random.random() < 0.3 else 
                                                 (0.05 + 0.25 * np.random.random()))) 
                   if row['status'] != 'Planejado' else 0,
        axis=1
    )
    
    # Data real/estimada de conclusão
    df_projetos['data_conclusao_real'] = df_projetos.apply(
        lambda row: row['data_conclusao_prevista'] + dt.timedelta(days=row['atraso']),
        axis=1
    )
    
    # Custo adicional (%)
    df_projetos['custo_adicional_pct'] = df_projetos.apply(
        lambda row: 0 if np.random.random() < 0.2 else 
                   np.random.randint(5, 30) if row['status'] != 'Planejado' else 0,
        axis=1
    )
    
    # Custo real
    df_projetos['custo_real'] = df_projetos.apply(
        lambda row: row['orcamento'] * (1 + row['custo_adicional_pct']/100),
        axis=1
    )
    
    # Satisfação do cliente (1-10)
    df_projetos['satisfacao_cliente'] = df_projetos.apply(
        lambda row: np.random.randint(7, 11) if row['atraso'] == 0 and row['custo_adicional_pct'] == 0 else
                   np.random.randint(5, 8) if row['atraso'] < 30 and row['custo_adicional_pct'] < 15 else
                   np.random.randint(3, 6),
        axis=1
    )
    
    # Eficiência (custo por m²)
    areas = {
        'Residencial': (80, 500),
        'Comercial': (200, 2000),
        'Industrial': (500, 10000),
        'Infraestrutura': (1000, 50000)
    }
    
    df_projetos['area'] = df_projetos.apply(
        lambda row: np.random.randint(areas[row['tipo_projeto']][0], areas[row['tipo_projeto']][1]),
        axis=1
    )
    
    df_projetos['eficiencia'] = df_projetos['custo_real'] / df_projetos['area']
    
    # Qualidade (1-100)
    df_projetos['qualidade'] = df_projetos.apply(
        lambda row: np.random.randint(85, 101) if row['complexidade'] in ['Baixa', 'Média'] else
                   np.random.randint(75, 96),
        axis=1
    )
    
    # Gerar dados de marcos do projeto
    marcos = []
    for _, projeto in df_projetos.iterrows():
        n_marcos = random.randint(4, 8)
        duracao_total = (projeto['data_conclusao_real'] - projeto['data_inicio']).days
        
        pontos = sorted([0] + [random.random() for _ in range(n_marcos-2)] + [1.0])
        
        for i, ponto in enumerate(pontos):
            data_marco = projeto['data_inicio'] + dt.timedelta(days=int(ponto * duracao_total))
            
            # Definir nome e tipo do marco
            if i == 0:
                nome_marco = "Início do Projeto"
                tipo_marco = "Início"
            elif i == len(pontos) - 1:
                nome_marco = "Entrega Final"
                tipo_marco = "Entrega"
            else:
                tipo_marco = random.choice(["Planejamento", "Execução", "Revisão", "Aprovação"])
                nome_marco = f"Fase {i} - {tipo_marco}"
            
            # Status do marco
            if data_marco <= dt.datetime.now():
                status_marco = "Concluído"
                atraso_marco = random.randint(-5, 15) if random.random() < 0.7 else random.randint(15, 45)
            else:
                status_marco = "Pendente"
                atraso_marco = 0
            
            marcos.append({
                'id_projeto': projeto['id_projeto'],
                'nome_marco': nome_marco,
                'tipo_marco': tipo_marco,
                'data_prevista': data_marco - dt.timedelta(days=atraso_marco if atraso_marco > 0 else 0),
                'data_real': data_marco if status_marco == "Concluído" else None,
                'status_marco': status_marco,
                'atraso_marco': atraso_marco if status_marco == "Concluído" else 0
            })
    
    df_marcos = pd.DataFrame(marcos)
    
    # Gerar dados financeiros mensais
    financeiros = []
    for _, projeto in df_projetos.iterrows():
        if projeto['status'] == 'Planejado':
            continue
            
        # Determinar o período de duração real
        inicio = projeto['data_inicio']
        fim = projeto['data_conclusao_real'] if projeto['status'] == 'Concluído' else dt.datetime.now()
        
        # Gerar dados mensais
        current_date = inicio.replace(day=1)
        end_month = fim.replace(day=28) + dt.timedelta(days=4)  # Garante que estamos no próximo mês
        
        # Calcular orçamento mensal (distribuição não uniforme)
        n_months = ((end_month.year - current_date.year) * 12 + 
                    end_month.month - current_date.month)
        
        # Criar distribuição de gasto típica (mais no meio do projeto)
        if n_months <= 1:
            monthly_weights = [1.0]
        else:
            x = np.linspace(-2, 2, n_months)
            monthly_weights = np.exp(-(x**2)/2)
            monthly_weights = monthly_weights / monthly_weights.sum()
        
        month_idx = 0
        while current_date < end_month:
            next_month = current_date.replace(day=28) + dt.timedelta(days=4)
            next_month = next_month.replace(day=1)
            
            # Calcular gastos orçados para o mês
            if n_months > 0:
                peso_mes = monthly_weights[min(month_idx, len(monthly_weights)-1)]
                orcado_mes = projeto['orcamento'] * peso_mes
            else:
                orcado_mes = projeto['orcamento']
            
            # Gasto real
            gasto_mes = orcado_mes * (1 + (projeto['custo_adicional_pct']/100) * 
                                     random.uniform(0.8, 1.2))
            
            financeiros.append({
                'id_projeto': projeto['id_projeto'],
                'data': current_date,
                'mes_ano': current_date.strftime("%m/%Y"),
                'orcado': orcado_mes,
                'realizado': gasto_mes,
                'variacao': (gasto_mes - orcado_mes) / orcado_mes * 100  # variação percentual
            })
            
            current_date = next_month
            month_idx += 1
    
    df_financeiro = pd.DataFrame(financeiros)
    
    # Salvar os datasets
    df_projetos.to_csv('data/projetos.csv', index=False)
    df_marcos.to_csv('data/marcos_projeto.csv', index=False)
    df_financeiro.to_csv('data/financeiro_mensal.csv', index=False)
    
    return df_projetos, df_marcos, df_financeiro

def gerar_dados_recursos(df_projetos):
    """
    Gera dados sobre recursos utilizados nos projetos.
    
    Parameters:
    -----------
    df_projetos : pd.DataFrame
        DataFrame com os dados dos projetos
        
    Returns:
    --------
    pd.DataFrame
        DataFrame com os dados de recursos
    """
    # Lista de recursos comuns
    categorias_recursos = [
        'Mão de obra especializada',
        'Mão de obra geral',
        'Materiais de construção',
        'Equipamentos',
        'Software e tecnologia',
        'Consultoria',
        'Licenciamentos',
        'Logística'
    ]
    
    # Gerar alocação de recursos por projeto
    recursos = []
    
    for _, projeto in df_projetos.iterrows():
        # Definir quantas categorias de recursos este projeto utiliza
        n_categorias = min(len(categorias_recursos), random.randint(3, len(categorias_recursos)))
        cats_projeto = random.sample(categorias_recursos, n_categorias)
        
        # Definir alocação percentual (somando 100%)
        alocacoes = np.random.dirichlet(np.ones(n_categorias)) * 100
        
        # Valores absolutos baseados no orçamento
        valores_absolutos = alocacoes * projeto['orcamento'] / 100
        
        for i, categoria in enumerate(cats_projeto):
            recursos.append({
                'id_projeto': projeto['id_projeto'],
                'categoria_recurso': categoria,
                'alocacao_percentual': alocacoes[i],
                'valor_orcado': valores_absolutos[i],
                'valor_realizado': valores_absolutos[i] * (1 + projeto['custo_adicional_pct']/100 * 
                                                        random.uniform(0.7, 1.3))
            })
    
    df_recursos = pd.DataFrame(recursos)
    df_recursos.to_csv('data/recursos_projeto.csv', index=False)
    
    return df_recursos

# Executar geração de dados
if __name__ == "__main__":
    # Verificar se o diretório de dados existe, caso contrário, criar

# KPI card creation function
def create_kpi_card(title, value, change=None, icon=None, color='primary'):
    """
    Create a KPI card component
    
    Parameters:
    -----------
    title : str
        Card title
    value : str
        Main KPI value
    change : str, optional
        Change indicator text
    icon : str, optional
        FontAwesome icon name
    color : str, optional
        Theme color key
        
    Returns:
    --------
    dbc.Card
        Formatted KPI card
    """
    card_content = [
        html.Div([
            html.Div([
                html.H4(title, className="card-title text-muted"),
                html.H2(value, className="card-value")
            ], className="col-9"),
            
            html.Div([
                html.I(className=f"fa fa-{icon} fa-3x text-{color}") if icon else "",
            ], className="col-3 text-right d-flex align-items-center justify-content-center")
        ], className="row")
    ]
    
    if change is not None:
        try:
            change_value = float(change.strip('%+-'))
            direction = "up" if change_value > 0 else "down"
            change_color = "success" if direction == "up" else "danger"
            
            card_content.append(
                html.Div([
                    html.I(className=f"fa fa-arrow-{direction} me-1"),
                    html.Span(change),
                    html.Span(" vs. período anterior", className="ms-1 small")
                ], className=f"mt-2 text-{change_color}")
            )
        except ValueError:
            # Handle case when change isn't a valid number
            pass
    
    return dbc.Card(card_content, className="kpi-card h-100 shadow-sm")

# Generate insights function
def generate_insights(filtered_df, filtered_financeiro=None):
    """
    Generate automated insights based on filtered data
    
    Parameters:
    -----------
    filtered_df : pd.DataFrame
        Filtered projects dataframe
    filtered_financeiro : pd.DataFrame, optional
        Filtered financial data dataframe
        
    Returns:
    --------
    list
        List of HTML components with insights
    """
    insights = []
    
    if filtered_df.empty:
        return [html.P("Não há dados para gerar insights com os filtros atuais.", className="text-muted")]
    
    # Insight 1: Critical projects
    critical_projects = filtered_df[
        (filtered_df['atraso'] > 30) & 
        (filtered_df['custo_adicional_pct'] > 15)
    ]
    if len(critical_projects) > 0:
        insights.append(html.Div([
            html.H6(f"{len(critical_projects)} Projetos Críticos Identificados", 
                   className="text-danger"),
            html.P(f"Projetos com atraso superior a 30 dias e custo adicional acima de 15%")
        ]))
    
    # Insight 2: Efficiency by project type
    if 'tipo_projeto' in filtered_df.columns and 'eficiencia' in filtered_df.columns:
        efficiency_by_type = filtered_df.groupby('tipo_projeto')['eficiencia'].mean().sort_values()
        if not efficiency_by_type.empty and len(efficiency_by_type) > 1:
            most_efficient = efficiency_by_type.index[0]
            least_efficient = efficiency_by_type.index[-1]
            
            insights.append(html.Div([
                html.H6("Eficiência por Tipo de Projeto", className="text-primary mt-3"),
                html.P([
                    f"Projetos do tipo ", 
                    html.Strong(most_efficient), 
                    f" apresentam maior eficiência de custo (R${efficiency_by_type[0]:.2f}/m²)",
                    html.Br(),
                    f"Projetos do tipo ",
                    html.Strong(least_efficient),
                    f" apresentam menor eficiência (R${efficiency_by_type[-1]:.2f}/m²)"
                ])
            ]))
    
    # Insight 3: Cost trend
    if filtered_financeiro is not None and len(filtered_financeiro) > 0:
        df_trend = filtered_financeiro.copy()
        df_trend['year_month'] = pd.to_datetime(df_trend['data']).dt.to_period('M')
        
        try:
            trend_data = df_trend.groupby('year_month').agg({
                'variacao': 'mean'
            }).reset_index()
            
            recent_months = min(3, len(trend_data))
            if recent_months > 0:
                recent_avg = trend_data['variacao'].tail(recent_months).mean()
                trend_direction = "crescente" if recent_avg > 2 else "decrescente" if recent_avg < -2 else "estável"
                
                insights.append(html.Div([
                    html.H6("Tendência Recente de Custos", className="text-info mt-3"),
                    html.P(f"Nos últimos {recent_months} meses, os custos mostram tendência {trend_direction} ({recent_avg:.1f}%)")
                ]))
        except Exception as e:
            insights.append(html.Div([
                html.H6("Análise de Tendência", className="text-info mt-3"),
                html.P(f"Não foi possível calcular a tendência com os dados atuais.")
            ]))
    
    # If no insights were generated
    if not insights:
        return [html.P("Ajuste os filtros para gerar insights baseados nos dados.", className="text-muted")]
    
    return insights

# Callbacks
@callback(
    [Output('portfolio-trend-chart', 'figure'),
     Output('cost-schedule-scatter', 'figure'),
     Output('tipo-chart', 'figure'),
     Output('region-performance-chart', 'figure'),
     Output('status-chart', 'figure'),
     Output('resources-chart', 'figure'),
     Output('efficiency-chart', 'figure'),
     Output('insights-content', 'children')],
    [Input('date-filter', 'start_date'),
     Input('date-filter', 'end_date'),
     Input('tipo-filter', 'value'),
     Input('regiao-filter', 'value')]
)
def update_charts(start_date, end_date, tipos, regioes):
    """
    Update all charts based on filter values
    """
    # Filter data
    filtered_df = df_projetos.copy()
    
    if start_date and end_date:
        filtered_df = filtered_df[(filtered_df['data_inicio'] >= start_date) & 
                                  (filtered_df['data_conclusao_prevista'] <= end_date)]
    
    if tipos and len(tipos) > 0:
        filtered_df = filtered_df[filtered_df['tipo_projeto'].isin(tipos)]
    
    if regioes and len(regioes) > 0:
        filtered_df = filtered_df[filtered_df['regiao'].isin(regioes)]
    
    # Filter related datasets
    filtered_marcos = df_marcos[df_marcos['id_projeto'].isin(filtered_df['id_projeto'])]
    filtered_financeiro = df_financeiro[df_financeiro['id_projeto'].isin(filtered_df['id_projeto'])]
    filtered_recursos = df_recursos[df_recursos['id_projeto'].isin(filtered_df['id_projeto'])]
    
    # Generate insights
    insights = generate_insights(filtered_df, filtered_financeiro)
    
    # 1. Portfolio Trend Chart
    # Aggregate data by month
    df_trend = filtered_financeiro.copy()
    
    # Create empty figure if no data
    if df_trend.empty:
        fig_trend = go.Figure()
        fig_trend.update_layout(
            template=TEMPLATE,
            title="Sem dados disponíveis para os filtros selecionados",
            xaxis=dict(title="Mês"),
            yaxis=dict(title="Valor (R$)"),
        )
    else:
        df_trend['year_month'] = pd.to_datetime(df_trend['data']).dt.to_period('M')
        df_trend_grouped = df_trend.groupby('year_month').agg(
            orcado=('orcado', 'sum'),
            realizado=('realizado', 'sum')
        ).reset_index()
        df_trend_grouped['year_month'] = df_trend_grouped['year_month'].dt.to_timestamp()
        
        fig_trend = go.Figure()
        
        fig_trend.add_trace(go.Scatter(
            x=df_trend_grouped['year_month'], 
            y=df_trend_grouped['orcado'],
            mode='lines+markers',
            name='Orçado',
            line=dict(color=COLOR_PALETTE['primary'], width=2),
            marker=dict(size=8, symbol='circle')
        ))
        
        fig_trend.add_trace(go.Scatter(
            x=df_trend_grouped['year_month'], 
            y=df_trend_grouped['realizado'],
            mode='lines+markers',
            name='Realizado',
            line=dict(color=COLOR_PALETTE['secondary'], width=2),
            marker=dict(size=8, symbol='circle')
        ))
        
        fig_trend.update_layout(
            template=TEMPLATE,
            margin=dict(l=40, r=20, t=20, b=20),
            legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1),
            xaxis=dict(title="Mês", titlefont=dict(size=AXIS_FONT_SIZE)),
            yaxis=dict(title="Valor (R$)", titlefont=dict(size=AXIS_FONT_SIZE)),
            hovermode='closest'
        )
    
    # 2. Cost vs Schedule Scatter
    if filtered_df.empty:
        fig_scatter = go.Figure()
        fig_scatter.update_layout(
            template=TEMPLATE,
            title="Sem dados disponíveis para os filtros selecionados",
            xaxis=dict(title="Atraso (dias)"),
            yaxis=dict(title="Custo Adicional (%)"),
        )
    else:
        fig_scatter = px.scatter(
            filtered_df,
            x='atraso',
            y='custo_adicional_pct',
            color='tipo_projeto',
            size='orcamento',
            hover_name='nome_projeto',
            hover_data={
                'orcamento': ":,.2f",  # Currency formatting
                'custo_real': ":,.2f",
                'atraso': True,
                'satisfacao_cliente': True,
                'regiao': True
            },
            color_discrete_map={
                'Residencial': COLOR_PALETTE['primary'],
                'Comercial': COLOR_PALETTE['secondary'],
                'Industrial': COLOR_PALETTE['warning'],
                'Infraestrutura': COLOR_PALETTE['info']
            },
            labels={
                'atraso': 'Atraso (dias)',
                'custo_adicional_pct': 'Custo Adicional (%)',
                'orcamento': 'Orçamento (R$)',
                'tipo_projeto': 'Tipo de Projeto'
            }
        )
        
        # Add quadrant lines
        fig_scatter.add_shape(
            type="line", x0=0, y0=0, x1=0, y1=30,
            line=dict(color="gray", width=1, dash="dash")
        )
        fig_scatter.add_shape(
            type="line", x0=0, y0=0, x1=180, y1=0,
            line=dict(color="gray", width=1, dash="dash")
        )
        
        # Add quadrant annotations
        fig_scatter.add_annotation(
            x=20, y=5, text="No Prazo<br>Acima do Orçamento",
            showarrow=False, font=dict(size=10, color="gray")
        )
        fig_scatter.add_annotation(
            x=-20, y=5, text="Adiantado<br>Acima do Orçamento",
            showarrow=False, font=dict(size=10, color="gray")
        )
        fig_scatter.add_annotation(
            x=20, y=-5, text="No Prazo<br>Abaixo do Orçamento",
            showarrow=False, font=dict(size=10, color="gray")
        )
        fig_scatter.add_annotation(
            x=-20, y=-5, text="Adiantado<br>Abaixo do Orçamento",
            showarrow=False, font=dict(size=10, color="gray")
        )
        
        fig_scatter.update_layout(
            template=TEMPLATE,
            margin=dict(l=40, r=20, t=20, b=20),
            legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1)
        )
    
    # 3. Projects by Type
    if filtered_df.empty:
        fig_tipo = go.Figure()
        fig_tipo.update_layout(
            template=TEMPLATE,
            title="Sem dados disponíveis para os filtros selecionados",
        )
    else:
        df_tipo = filtered_df.groupby('tipo_projeto').agg(
            count=('id_projeto', 'count'),
            orcamento_total=('orcamento', 'sum')
        ).reset_index()
        
        fig_tipo = px.pie(
            df_tipo,
            values='orcamento_total',
            names='tipo_projeto',
            hole=0.5,
            color_discrete_sequence=[
                COLOR_PALETTE['primary'], 
                COLOR_PALETTE['secondary'],
                COLOR_PALETTE['warning'],
                COLOR_PALETTE['info']
            ]
        )
        
        fig_tipo.update_traces(
            textposition='outside',
            textinfo='percent+label',
            marker=dict(line=dict(color='white', width=2))
        )
        
        fig_tipo.update_layout(
            template=TEMPLATE,
            margin=dict(l=20, r=20, t=20, b=20),
            legend=dict(
                orientation="h",
                yanchor="bottom",
                y=-0.15,
                xanchor="center",
                x=0.5
            ),
            annotations=[
                dict(
                    text=f"{len(filtered_df)}<br>Projetos",
                    x=0.5, y=0.5,
                    font_size=16,
                    font_color=COLOR_PALETTE['dark'],
                    showarrow=False
                )
            ]
        )
    
    # 4. Regional Performance
    if filtered_df.empty:
        fig_region = go.Figure()
        fig_region.update_layout(
            template=TEMPLATE,
            title="Sem dados disponíveis para os filtros selecionados",
            xaxis=dict(title="Região"),
            yaxis=dict(title="Número de Projetos"),
        )
    else:
        df_region = filtered_df.groupby('regiao').agg(
            count=('id_projeto', 'count'),
            atraso_medio=('atraso', 'mean'),
            custo_adicional_medio=('custo_adicional_pct', 'mean'),
            satisfacao_media=('satisfacao_cliente', 'mean')
        ).reset_index()
        
        fig_region = go.Figure()
        
        fig_region.add_trace(go.Bar(
            x=df_region['regiao'],
            y=df_region['count'],
            name='Número de Projetos',
            marker_color=COLOR_PALETTE['primary'],
            opacity=0.8
        ))
        
        # Create secondary y-axis
        fig_region.add_trace(go.Scatter(
            x=df_region['regiao'], 
            y=df_region['satisfacao_media'],
            mode='markers',
            name='Satisfação Média',
            marker=dict(
                color=COLOR_PALETTE['warning'],
                size=12,
                symbol='diamond',
                line=dict(color='white', width=1)
            ),
            yaxis='y2'
        ))
        
        fig_region.update_layout(
            template=TEMPLATE,
            margin=dict(l=40, r=40, t=20, b=20),
            legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1),
            yaxis=dict(
                title="Projetos",
                titlefont=dict(color=COLOR_PALETTE['primary']),
                tickfont=dict(color=COLOR_PALETTE['primary'])
            ),
            yaxis2=dict(
                title="Satisfação (0-10)",
                titlefont=dict(color=COLOR_PALETTE['warning']),
                tickfont=dict(color=COLOR_PALETTE['warning']),
                anchor="x",
                overlaying="y",
                side="right",
                range=[0, 10]
            )
        )
    
    # 5. Status Chart
    if filtered_df.empty:
        fig_status = go.Figure()
        fig_status.update_layout(
            template=TEMPLATE,
            title="Sem dados disponíveis para os filtros selecionados",
            yaxis=dict(title="Status"),
            xaxis=dict(title="Número de Projetos"),
        )
    else:
        df_status = filtered_df.groupby('status').agg(
            count=('id_projeto', 'count'),
        ).reset_index()
        
        fig_status = go.Figure()
        
        fig_status.add_trace(go.Bar(
            y=df_status['status'],
            x=df_status['count'],
            orientation='h',
            marker=dict(
                color=[
                    COLOR_PALETTE['success'] if status == 'Concluído' else
                    COLOR_PALETTE['warning'] if status == 'Em Andamento' else
                    COLOR_PALETTE['info']
                    for status in df_status['status']
                ]
            ),
            text=df_status['count'],
            textposition='auto'
        ))
        
        fig_status.update_layout(
            template=TEMPLATE,
            margin=dict(l=20, r=20, t=20, b=20),
            xaxis_title="Número de Projetos",
            yaxis=dict(
                categoryorder='array',
                categoryarray=['Planejado', 'Em Andamento', 'Concluído']
            ),
            showlegend=False
        )
    
    # 6. Resource Distribution
    if filtered_recursos.empty:
        fig_resources = go.Figure()
        fig_resources.update_layout(
            template=TEMPLATE,
            title="Sem dados disponíveis para os filtros selecionados",
            xaxis=dict(title="Categoria de Recurso"),
            yaxis=dict(title="Valor (R$)"),
        )
    else:
        recursos_agg = filtered_recursos.groupby('categoria_recurso').agg(
            valor_orcado=('valor_orcado', 'sum'),
            valor_realizado=('valor_realizado', 'sum')
        ).reset_index()
        
        recursos_agg = recursos_agg.sort_values('valor_orcado', ascending=False)
        
        fig_resources = go.Figure()
        
        fig_resources.add_trace(go.Bar(
            x=recursos_agg['categoria_recurso'],
            y=recursos_agg['valor_orcado'],
            name='Orçado',
            marker_color=COLOR_PALETTE['primary'],
            opacity=0.7
        ))
        
        fig_resources.add_trace(go.Bar(
            x=recursos_agg['categoria_recurso'],
            y=recursos_agg['valor_realizado'],
            name='Realizado',
            marker_color=COLOR_PALETTE['secondary'],
            opacity=0.7
        ))
        
        fig_resources.update_layout(
            template=TEMPLATE,
            margin=dict(l=40, r=20, t=20, b=80),

            barmode='group',
            xaxis=dict(
                title="Categoria de Recurso",
                titlefont=dict(size=AXIS_FONT_SIZE),
                tickfont=dict(size=10),
                tickangle=45
            ),
            yaxis=dict(title="Valor (R$)", titlefont=dict(size=AXIS_FONT_SIZE))
        )
    
    # 7. Efficiency Chart (cost per m²)
    if filtered_df.empty:
        fig_efficiency = go.Figure()
        fig_efficiency.update_layout(
            template=TEMPLATE,

            xaxis=dict(title="Tipo de Projeto"),
            yaxis=dict(title="Custo por m² (R$)"),
        )
    else:
        df_efficiency = filtered_df.copy()

            df_efficiency,

            template=TEMPLATE,
            margin=dict(l=40, r=20, t=20, b=20),
            legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1),
            showlegend=False
        )
    
    return fig_trend, fig_scatter, fig_tipo, fig_region, fig_status, fig_resources, fig_efficiency, insights
