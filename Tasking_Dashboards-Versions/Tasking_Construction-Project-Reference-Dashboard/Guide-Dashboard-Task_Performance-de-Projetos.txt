O dashboard desenvolvido oferece uma análise abrangente de performance de projetos com:

- **Múltiplas visualizações** organizadas em hierarquia lógica
- **Design responsivo** com cards e elementos visuais bem definidos

```python
def generate_insights(start_date, end_date, tipos, regioes):
    # Filtrar dados como no callback principal
    filtered_df = df_projetos.copy()
    # (código de filtragem)
    
    # Gerar insights baseados nos dados filtrados
    insights = []
    
    # Insight 1: Projetos críticos
    critical_projects = filtered_df[
        (filtered_df['atraso'] > 30) & 
        (filtered_df['custo_adicional_pct'] > 15)
    ]
    if len(critical_projects) > 0:
        insights.append(html.Div([
            html.H6(f"{len(critical_projects)} Projetos Críticos Identificados", 
                   className="text-danger"),
            html.P(f"Projetos com atraso superior a 30 dias e custo adicional acima de 15%")
        ]))
    
    # Insight 2: Eficiência por tipo
    efficiency_by_type = filtered_df.groupby('tipo_projeto')['eficiencia'].mean().sort_values()
    most_efficient = efficiency_by_type.index[0]
    least_efficient = efficiency_by_type.index[-1]
    
    insights.append(html.Div([
        html.H6("Eficiência por Tipo de Projeto", className="text-primary mt-3"),
        html.P([
            f"Projetos do tipo ", 
            html.Strong(most_efficient), 
            f" apresentam maior eficiência de custo (R${efficiency_by_type[0]:.2f}/m²)",
            html.Br(),
            f"Projetos do tipo ",
            html.Strong(least_efficient),
            f" apresentam menor eficiência (R${efficiency_by_type[-1]:.2f}/m²)"
        ])
    ]))
    
    # Insight 3: Tendência temporal
    if filtered_financeiro is not None and len(filtered_financeiro) > 0:
        df_trend = filtered_financeiro.copy()
        df_trend['year_month'] = pd.to_datetime(df_trend['data']).dt.to_period('M')
        trend_data = df_trend.groupby('year_month').agg({
            'variacao': 'mean'
        }).reset_index()
        
        recent_months = min(3, len(trend_data))
        if recent_months > 0:
            recent_avg = trend_data['variacao'].tail(recent_months).mean()
            trend_direction = "crescente" if recent_avg > 2 else "decrescente" if recent_avg < -2 else "estável"
            
            insights.append(html.Div([
                html.H6("Tendência Recente de Custos", className="text-info mt-3"),
                html.P(f"Nos últimos {recent_months} meses, os custos mostram tendência {trend_direction} ({recent_avg:.1f}%)")
            ]))
    
    # Se não houver insights devido a filtros muito restritivos
    if not insights:
        return html.P("Ajuste os filtros para gerar insights baseados nos dados.", className="text-muted")
    
    return insights
```

### 6. Exportação do HTML Final

Para gerar o arquivo HTML final para submissão:

```python
# No final do viz.py, adicionar:
if __name__ == '__main__':
    # Código existente para gerar dados se necessário
    
    # Para executar o servidor durante desenvolvimento
    if 'debug' in sys.argv:
        app.run_server(debug=True)
    else:
        # Para gerar o arquivo HTML final
        from dash.dash import _generate_index_html
        
        # Gerar HTML completo do dashboard
        html_str = _generate_index_html(
            app.config.title,
            app.config.metas,
            app.config.favicon,
            app.config.css,
            app._generate_scripts_html(),
            app.config.serve_locally
        )
        
        # Substituir pelo index_string personalizado
        html_str = app.index_string
        
        # Garantir que o diretório de saída exista
        if not os.path.exists('outputs'):
            os.makedirs('outputs')
        
        # Salvar arquivo HTML final
        with open('outputs/dashboard.html', 'w', encoding='utf-8') as f:
            f.write(html_str)
            
        print("Dashboard HTML gerado com sucesso em outputs/dashboard.html")
```
